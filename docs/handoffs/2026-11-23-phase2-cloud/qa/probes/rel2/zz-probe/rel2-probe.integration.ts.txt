async function measure(page: Page, axePath: string) {
  const layout = await page.evaluate(() => {
    const de = document.documentElement;
    const vis = (el: Element) => { const r = el.getBoundingClientRect(); const cs = getComputedStyle(el); return r.width > 0 && r.height > 0 && cs.visibility !== 'hidden' && cs.display !== 'none' && !el.closest('[aria-hidden="true"],[hidden],[inert]') && ((el as any).checkVisibility?.({ checkOpacity: true, checkVisibilityCSS: true }) ?? true); };
    const leaves = Array.from(document.body.querySelectorAll('*')).filter(el => vis(el) && Array.from(el.childNodes).some(n => n.nodeType === 3 && n.textContent!.trim().length > 1));
    const clipped: string[] = [];
    for (const el of leaves) { const cs = getComputedStyle(el), h = el as HTMLElement; if ((cs.textOverflow === 'ellipsis' || ['hidden', 'clip'].includes(cs.overflowX) || ['hidden', 'clip'].includes(cs.overflowY)) && (h.scrollWidth > h.clientWidth + 1 || h.scrollHeight > h.clientHeight + 2) && h.clientWidth > 4 && h.clientHeight > 4 && !el.closest('.sr-only,.visually-hidden,.b-sr,[class*="sr-only"]')) clipped.push(h.innerText.trim().slice(0, 40)); }
    return { overflowX: de.scrollWidth - de.clientWidth, clipped, rem: getComputedStyle(de).fontSize };
  });
  await page.addScriptTag({ path: axePath });
  const axe = await page.evaluate(async () => { const r = await (window as any).axe.run(document, { runOnly: { type: 'tag', values: ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'] }, resultTypes: ['violations'] }); return r.violations.map((v: any) => ({ id: v.id, impact: v.impact, count: v.nodes.length, targets: v.nodes.slice(0, 3).map((n: any) => n.target.join(' ')) })); });
  return { ...layout, axe };
}

async function signIn(page: Page, base: string, id: string) {
  await page.goto(base + '/login');
  await page.getByLabel('Employee ID', { exact: true }).fill(id);
  await page.getByLabel('Password', { exact: true }).fill(TEST_PASSWORD);
  await page.getByRole('button', { name: 'Sign in', exact: true }).click();
}
