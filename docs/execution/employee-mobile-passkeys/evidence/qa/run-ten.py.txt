from pathlib import Path
import subprocess,json,datetime,os,sys,hashlib
repo=Path(sys.argv[1]);dest=Path(sys.argv[2]);dest.mkdir(parents=True,exist_ok=True)
files=json.loads(Path(sys.argv[3]).read_text())
env={k:v for k,v in os.environ.items() if k in ['PATH','HOME','TMPDIR','LANG','LC_ALL']};env['NODE_ENV']='test'
summary=[]
for i,f in enumerate(files):
 out=dest/(str(i+1).zfill(2)+'-'+Path(f).name);out.mkdir(exist_ok=True)
 argv=['node','node_modules/vitest/vitest.mjs','run',f,'--maxWorkers=1','--no-file-parallelism','--reporter=default','--reporter=json','--outputFile='+str(out/'results.json')]
 invocation={'at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'cwd':str(repo),'argv':argv,'environment':env,'candidate':subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip(),'testSHA256':hashlib.sha256((repo/f).read_bytes()).hexdigest()}
 (out/'invocation.json').write_text(json.dumps(invocation,indent=2))
 with (out/'stdout.log').open('w') as log:r=subprocess.run(argv,cwd=repo,env=env,stdout=log,stderr=subprocess.STDOUT)
 row={'file':f,'exitCode':r.returncode,'artifacts':str(out)}
 if (out/'results.json').exists():
  results=json.loads((out/'results.json').read_text());row.update({k:results.get(k) for k in ['success','numTotalTests','numPassedTests','numFailedTests','numPendingTests']})
  row['failures']=[{'name':a['fullName'],'messages':a.get('failureMessages',[])} for t in results['testResults'] for a in t['assertionResults'] if a['status']=='failed']
 summary.append(row);(dest/'summary.json').write_text(json.dumps(summary,indent=2));print(json.dumps({k:v for k,v in row.items() if k!='failures'}),flush=True)
