  17 function EmployeePasskeyDesktop(props:ViewProps){
  18  const dialog=useRef<HTMLDialogElement>(null),fallbackAdd=useRef<HTMLButtonElement>(null),{state,confirm}=props;
  19  useEffect(()=>{if(confirm){dialog.current?.showModal();dialog.current?.querySelector<HTMLElement>('[data-autofocus]')?.focus();}else if(dialog.current?.open)dialog.current.close();},[confirm]);
  20  return <div data-passkey-state={state.phase}><PasskeyDesktopView load={state.phase==='ready'?'ready':state.phase==='unavailable'||state.phase==='denied'?'failed':'loading'} passkeys={state.phase==='ready'?state.passkeys??[]:[]} busy={state.busy==='removing'?'remove':state.busy?'add':null} notice={state.notice?{tone:/^(Passkey added|Passkey removed)/.test(state.notice)?'done':'error',text:state.notice}:null} confirm={state.phase==='ready'?confirm:null} supported={props.supported} phone={false} disabled={state.phase!=='ready'||state.needsRefresh} dialog={dialog} addButton={props.addRef??fallbackAdd} onRefresh={props.onRefresh} onAdd={props.onAdd} onReview={props.onReview} onRemove={key=>props.onRemove(key.id)} onKeep={props.onKeep}/></div>;

  23 export function EmployeePasskeyView({mobile,state,supported,confirm,onAdd,onRefresh,onRemove,onReview,onKeep,onCancel,addRef,confirmationRef,feedbackRef}:ViewProps){
  24  const ready=state.phase==='ready',keys=ready?state.passkeys??[]:[],busy=state.busy!==null;
  25  const reviewed=ready?keys.find(key=>key.id===confirm?.id):null;
  26  if(state.phase==='disabled'||state.phase==='ended')return null;
  27  if(!mobile)return <EmployeePasskeyDesktop mobile={mobile} state={state} supported={supported} confirm={confirm} onAdd={onAdd} onRefresh={onRefresh} onRemove={onRemove} onReview={onReview} onKeep={onKeep} onCancel={onCancel} addRef={addRef}/>;
  28  return <section className={`nv-card pk-section em-passkeys${mobile?' em-passkeys-mobile':''}`} aria-labelledby="passkeys-title" data-passkey-state={state.phase}>
  29   <div className="nv-card__head"><h2 className="nv-card__title" id="passkeys-title">{mobile?'Passkeys':'Face ID and passkeys'}</h2></div>
  30   <div className="nv-card__body pk-body">
  31    <p className="pk-muted em-passkeys-intro">Sign in with your device’s screen lock instead of typing your password.</p>
  32    {state.phase==='loading'?<p className="pk-muted" role="status">Checking your saved passkeys…</p>:null}
  33    {state.phase==='paused'&&!busy?<p className="pk-muted" role="status">Passkeys are paused. Return to this page to check your account.</p>:null}
  34    {state.phase==='denied'?<p className="pk-muted" role="status">Your session changed. <a href="/login">Sign in again</a> to manage your passkeys.</p>:null}
  35    {state.phase==='unavailable'?<p className="pk-muted">{state.needsRefresh?'Check your saved passkeys before starting another request.':'We couldn’t check your passkeys. Refresh to try again.'}</p>:null}

  37     {keys.length?<ul className="pk-list">{keys.map(key=><li className="pk-item" key={key.id}>

  40      <button type="button" className="nv-link pk-remove" aria-label={'Remove '+key.name} disabled={busy||state.needsRefresh||!!reviewed} onClick={event=>onReview(key,event.currentTarget)}>Remove</button>
  41     </li>)}</ul>:<div className="em-passkeys-empty"><KeyIcon size={28} aria-hidden="true"/><div><h3>No saved passkeys</h3><p className="pk-muted">Your password still works. Add a passkey when you’re ready.</p></div></div>}

  45      <div className="em-passkeys-actions"><button type="button" className="nv-btn" data-kind="secondary" disabled={busy} onClick={onKeep}>Keep passkey</button><button type="button" className="nv-btn" data-kind="danger" disabled={busy} onClick={()=>onRemove(reviewed.id)}>Remove passkey</button></div>
  46     </div>:null}

  49    </>:null}
  50    {busy?<div className="em-passkeys-progress"><p role="status">{state.busy==='device'?'Finish setup in your device’s prompt.':state.busy==='removing'?'Removing your passkey…':state.busy==='verifying'?'Checking and saving your passkey…':'Preparing secure setup…'}</p>{state.busy==='device'||state.busy==='options'?<button type="button" className="nv-btn" data-kind="secondary" onClick={onCancel}>Cancel setup</button>:null}</div>:null}

  53    <details className="em-passkeys-help"><summary><span>How passkeys work</span><CaretDownIcon size={20} aria-hidden="true"/></summary><div><p>Your device may use a fingerprint, face scan, screen-lock code or security key.   54   </div>

  58 /** Stable lifetime; responsive changes replace presentation, never a ceremony. */
  59 export function EmployeePasskeySection({authority}:{authority:string}){
  60  const mobile=useEmployeeMobile(),flag=passkeysFlagOn();
  61  const supported=useSyncExternalStore(subscribe,()=>passkeysAvailable()&&typeof navigator.credentials?.create==='function'&&window.isSecureContext,()=>false);
  62  const [state,setState]=useState<PasskeyState>({phase:'loading',busy:null,passkeys:null,notice:'',needsRefresh:false});
  63  const [confirm,setConfirm]=useState<Passkey|null>(null),owner=useRef<ReturnType<typeof createPasskeyManager>|null>(null);
  64  const add=useRef<HTMLButtonElement>(null),confirmation=useRef<HTMLDivElement>(null),feedback=useRef<HTMLParagraphElement>(null),opener=useRef<HTMLButtonElement|null>(null),completion=useRef<'add'|'remove'|null>(null);
  65  useEffect(()=>{
  66   if(!flag)return;
  67   const manager=createPasskeyManager(authority,()=>{if(manager.state.phase!=='ready')setConfirm(null);setState({...manager.state});});owner.current=manager;
  68   const resume=()=>{if(document.visibilityState!=='hidden'&&navigator.onLine!==false)void manager.resume();};
  69   const blur=()=>manager.pause(true),pause=()=>manager.pause(),end=()=>manager.end();
  70   const visibility=()=>document.visibilityState==='hidden'?pause():resume();
  71   const storage=(event:StorageEvent)=>{if(event.key===null||event.key==='fuel-portal-account-fence'||event.key==='fuel-portal-owner-generation')end();};
  72   void manager.refresh();window.addEventListener('blur',blur);window.addEventListener('pagehide',pause);window.addEventListener('offline',pause);window.addEventListener('focus',resume);window.addEventListener('pageshow',resume);window.addEventListener('online',resume);window.addEventListener('nevora:signout',end);window.addEventListener('storage',storage);document.addEventListener('visibilitychange',visibility);
  73   const poll=setInterval(()=>{if(document.visibilityState!=='hidden'&&manager.state.phase==='ready'&&!manager.state.busy)void manager.refresh();},30000);
  74   return()=>{clearInterval(poll);manager.end();owner.current=null;window.removeEventListener('blur',blur);window.removeEventListener('pagehide',pause);window.removeEventListener('offline',pause);window.removeEventListener('focus',resume);window.removeEventListener('pageshow',resume);window.removeEventListener('online',resume);window.removeEventListener('nevora:signout',end);window.removeEventListener('storage',storage);document.removeEventListener('visibilitychange',visibility);};
  75  },[authority,flag]);
  76  useEffect(()=>{if(confirm){confirmation.current?.focus();confirmation.current?.scrollIntoView({block:'nearest'});}},[confirm]);
  77  useEffect(()=>{
  78   if(!completion.current||state.busy||!state.notice||state.phase==='loading'||state.phase==='paused')return;
  79   completion.current=null;const cancelled=/^Setup cancelled\./.test(state.notice);
  80   const frame=requestAnimationFrame(()=>{if(owner.current?.state.phase==='ended')return;const target=cancelled||!mobile?add.current:feedback.current;target?.focus();target?.scrollIntoView({block:'center'});});return()=>cancelAnimationFrame(frame);
  81  },[mobile,state.busy,state.notice,state.phase]);
  82  if(!flag)return null;
  83  return <div data-passkey-owner=""><EmployeePasskeyView mobile={mobile} state={state} supported={supported} confirm={confirm} addRef={add} confirmationRef={confirmation} feedbackRef={feedback}
  84   onAdd={()=>{completion.current='add';void owner.current?.add();}} onRefresh={()=>void owner.current?.refresh()} onRemove={id=>{completion.current='remove';void owner.current?.remove(id);}} onCancel={()=>{completion.current='add';owner.current?.cancel();}}
  85   onReview={(key,button)=>{opener.current=button;setConfirm(key);}} onKeep={()=>{setConfirm(null);requestAnimationFrame(()=>{if(opener.current?.isConnected)opener.current.focus();else add.current?.focus();});}}/></div>;
  86 }
