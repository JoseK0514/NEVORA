  10 function parseKey(value:unknown):Passkey {
  11  if(!record(value)||!exact(value,['id','name','createdAt','lastUsedAt'])||typeof value.id!=='string'||!uuid.test(value.id)||typeof value.name!=='string'||!value.name.trim()||value.name.length>80||/[\u0000-\u001f\u007f]/.test(value.name)||typeof value.createdAt!=='string'||!Number.isFinite(Date.parse(value.createdAt))||value.lastUsedAt!==null&&(typeof value.lastUsedAt!=='string'||!Number.isFinite(Date.parse(value.lastUsedAt))))throw Error('Invalid passkey metadata');

  14 export function parsePasskeySnapshot(value:unknown,authority:string):Snapshot {
  15  if(!record(value)||!exact(value,['authority','validForMs','passkeys'])||value.authority!==authority||typeof value.validForMs!=='number'||!Number.isInteger(value.validForMs)||value.validForMs<=0||value.validForMs>45000||!Array.isArray(value.passkeys))throw Error('Invalid passkey snapshot');

  26 export function createPasskeyManager(authority:string,changed:()=>void){
  27  let state:PasskeyState={phase:'loading',busy:null,passkeys:null,notice:'',needsRefresh:false};
  28  let version=0,ended=false,deadline=0,expiry:ReturnType<typeof setTimeout>|undefined;
  29  let active:AbortController|undefined,reading=false;
  30  const publish=(next:Partial<PasskeyState>)=>{state={...state,...next};changed();};
  31  const clearLease=()=>{clearTimeout(expiry);deadline=0;};
  32  const expire=()=>{clearLease();publish({phase:'unavailable',passkeys:null,notice:'Refresh to check your current passkeys.'});};
  33  const valid=()=>{if(deadline<=performance.now()){expire();return false;}return state.phase==='ready'&&!state.busy&&!state.needsRefresh;};
  34  const isLive=(v:number)=>!ended&&v===version;
  35  function invalidate(){version++;active?.abort();active=undefined;reading=false;clearLease();}
  36  async function request(path:string,method:string,signal:AbortSignal,body?:unknown):Promise<unknown>{
  37   const controller=new AbortController();let timer:ReturnType<typeof setTimeout>|undefined;
  38   const abort=()=>controller.abort();signal.addEventListener('abort',abort,{once:true});if(signal.aborted)controller.abort();
  39   const stopped=new Promise<never>((_,reject)=>{controller.signal.addEventListener('abort',()=>reject(Error('Request stopped')),{once:true});if(controller.signal.aborted)reject(Error('Request stopped'));});
  40   timer=setTimeout(()=>controller.abort(),10000);
  41   try{return await Promise.race([stopped,(async()=>{
  42    const response=await fetch(path,{method,credentials:'same-origin',cache:'no-store',signal:controller.signal,headers:{'Content-Type':'application/json','X-Nevora-Passkey-Authority':authority},...(body===undefined?{}:{body:JSON.stringify(body)})});
  43    if(controller.signal.aborted){await response.body?.cancel().catch(()=>{});throw Error('Late reply');}
  44    if(response.status===401||response.status===403){await response.body?.cancel();throw new Denied();}
  45    if(response.status===204&&method==='DELETE')return null;
  46    if(response.ok&&(method==='DELETE'||response.status!==200)){await response.body?.cancel();throw Error('Unexpected receipt');}
  47    const value:unknown=await response.json();
  48    if(!response.ok){if(record(value)&&value.error==='not_supported')throw new Disabled();throw new Refused();}
  49    return value;
  50   })()]);}finally{clearTimeout(timer);signal.removeEventListener('abort',abort);}
  51  }
  52  function failure(error:unknown){clearLease();publish({phase:error instanceof Denied?'denied':error instanceof Disabled?'disabled':'unavailable',passkeys:null,busy:null});}
  53  async function read(v:number,signal:AbortSignal):Promise<boolean>{
  54   clearLease();publish({phase:'loading',passkeys:null});const started=performance.now();
  55   try{
  56    const result=parsePasskeySnapshot(await request('/api/auth/passkeys','GET',signal),authority);
  57    if(!isLive(v))return false;
  58    deadline=started+result.validForMs;const remaining=deadline-performance.now();
  59    if(remaining<=0){expire();return false;}
  60    publish({phase:'ready',passkeys:result.passkeys,needsRefresh:false});expiry=setTimeout(expire,remaining);return true;
  61   }catch(error){if(isLive(v))failure(error);return false;}
  62  }
  63  async function refresh(){
  64   if(ended||state.busy||reading||state.phase==='denied'||state.phase==='disabled')return;
  65   invalidate();reading=true;const v=version,controller=active=new AbortController();
  66   await read(v,controller.signal);if(isLive(v))reading=false;
  67  }
  68  async function afterMutation(v:number,signal:AbortSignal,check:(keys:Passkey[])=>boolean,notice:string){
  69   const current=await read(v,signal);if(!isLive(v))return;
  70   if(current)publish({notice:check(state.passkeys??[])?notice:uncertain,busy:null});
  71   else if(state.phase!=='denied'&&state.phase!=='disabled')publish({notice:uncertain,needsRefresh:true,busy:null});

  73  async function add(){
  74   if(ended||!valid())return;
  75   const v=version,controller=active=new AbortController();publish({busy:'options',notice:''});let sent=false;
  76   try{
  77    const options=await request('/api/auth/passkey/register/options','POST',controller.signal,{});
  78    if(!isLive(v))return;if(deadline<=performance.now()){expire();publish({busy:null});return;}
  79    if(!record(options))throw Error('Invalid options');publish({busy:'device'});
  80    // Blur from an OS credential sheet is expected. Explicit cancellation,
  81    // navigation, offline and signout still abort/fence this single ceremony.
  82    let nativeTimer:ReturnType<typeof setTimeout>|undefined;
  83    const credential=await Promise.race([
  84     navigator.credentials.create({publicKey:creationOptions(options),signal:controller.signal}),
  85     new Promise<never>((_,reject)=>{controller.signal.addEventListener('abort',()=>reject(new DOMException('Cancelled','AbortError')),{once:true});nativeTimer=setTimeout(()=>{controller.abort();reject(new DOMException('Cancelled','AbortError'));},300000);}),
  86    ]).finally(()=>clearTimeout(nativeTimer));
  87    if(!isLive(v))return;
  88    if(!credential){publish({busy:null});await read(v,controller.signal);if(isLive(v))publish({notice:'Setup cancelled. No passkey was added to your account.'});return;}
  89    publish({busy:'verifying'});
  90    if(!await read(v,controller.signal)||!isLive(v))return;
  91    if(deadline<=performance.now()){expire();publish({busy:null});return;}
  92    sent=true;
  93    const receipt=await request('/api/auth/passkey/register/verify','POST',controller.signal,{response:credentialJSON(credential as PublicKeyCredential),name:deviceName()});
  94    if(!isLive(v))return;
  95    if(!record(receipt)||!exact(receipt,['passkey']))throw Error('Invalid receipt');const key=parseKey(receipt.passkey);
  96    await afterMutation(v,controller.signal,keys=>keys.some(saved=>saved.id===key.id),'Passkey added. You can use it the next time you sign in.');
  97   }catch(error){
  98    if(!isLive(v))return;
  99    if(error instanceof Denied||error instanceof Disabled){failure(error);return;}
 100    if(sent){clearLease();publish({phase:'unavailable',passkeys:null,busy:null,needsRefresh:true,notice:uncertain});return;}
 101    publish({busy:null});await read(v,controller.signal.aborted?(active=new AbortController()).signal:controller.signal);
 102    if(isLive(v))publish({notice:error instanceof DOMException&&['AbortError','NotAllowedError'].includes(error.name)?'Setup cancelled. No passkey was added to your account.':'Setup did not finish. Check your connection and try again.'});
 103   }finally{if(isLive(v)&&state.busy)publish({busy:null});}
 104  }
 105  async function remove(id:string){
 106   if(ended||!valid()||!state.passkeys?.some(key=>key.id===id))return;
 107   const v=version,controller=active=new AbortController();publish({busy:'removing',notice:''});
 108   try{
 109    await request('/api/auth/passkeys/'+encodeURIComponent(id),'DELETE',controller.signal);
 110    if(isLive(v))await afterMutation(v,controller.signal,keys=>!keys.some(key=>key.id===id),'Passkey removed.');
 111   }catch(error){if(isLive(v)){if(error instanceof Denied||error instanceof Disabled)failure(error);else{clearLease();publish({phase:'unavailable',passkeys:null,busy:null,needsRefresh:true,notice:uncertain});}}}
 112   finally{if(isLive(v)&&state.busy)publish({busy:null});}

 114  function pause(expectedNativeBlur=false){
 115   if(ended)return;
 116   if(expectedNativeBlur&&state.busy==='device'){clearLease();publish({phase:'paused',passkeys:null});return;}
 117   const unknown=state.busy==='verifying'||state.busy==='removing';invalidate();publish({phase:'paused',passkeys:null,busy:null,needsRefresh:state.needsRefresh||unknown,notice:unknown?uncertain:state.notice});
 118  }
 119  async function resume(){if(!ended&&state.busy!=='device')await refresh();}
 120  function cancel(){if(state.busy!=='device'&&state.busy!=='options')return;pause();publish({notice:'Setup cancelled. No passkey was added to your account.'});void refresh();}
 121  function end(){if(ended)return;ended=true;invalidate();publish({phase:'ended',passkeys:null,busy:null,notice:'',needsRefresh:false});}
 122  return {get state(){return state;},refresh,add,remove,pause,resume,cancel,end};
