src/components/passkeys/passkey-section.tsx
   1 "use client";
   2 
   3 import { useCallback, useEffect, useRef, useState, useSyncExternalStore } from "react";
   4 import { PASSKEY_COPY } from "@/lib/passkeys/copy";
   5 import {PasskeyDesktopView} from "./passkey-desktop-view";
   6 import { deviceName, isPhoneLike, listPasskeys, passkeysAvailable, passkeysFlagOn, registerPasskey, removePasskey, type Passkey } from "@/lib/passkeys/client";
   7 import "./passkeys.css";
   8 
   9 const t = PASSKEY_COPY.en;
  10 const subscribe = () => () => {};
  11 
  12 type Load = { state: "loading" } | { state: "ready"; passkeys: Passkey[] } | { state: "failed" };
  13 
  14 /** Profile, every role: "Face ID and passkeys". Set up (named after the device), the saved list with added and
  15  * last-used dates, and Remove with a confirmation. Nothing renders while NEXT_PUBLIC_NEVORA_PASSKEYS is off. */
  16 export function PasskeySection() {
  17   const supported = useSyncExternalStore(subscribe, passkeysAvailable, () => false);
  18   const phone = useSyncExternalStore(subscribe, isPhoneLike, () => false);
  19   const [load, setLoad] = useState<Load>({ state: "loading" });
  20   const [busy, setBusy] = useState<"add" | "remove" | null>(null);
  21   const [notice, setNotice] = useState<{ tone: "done" | "error"; text: string } | null>(null);
  22   const [confirm, setConfirm] = useState<Passkey | null>(null);
  23   const dialog = useRef<HTMLDialogElement>(null);
  24   const opener = useRef<HTMLElement | null>(null);
  25   const addButton = useRef<HTMLButtonElement>(null);
  26   const flagOn = passkeysFlagOn();
  27 
  28   const refresh = useCallback(async () => {
  29     setLoad({ state: "loading" });
  30     const result = await listPasskeys();
  31     setLoad(result.ok ? { state: "ready", passkeys: result.value.passkeys ?? [] } : { state: "failed" });
  32   }, []);
  33   useEffect(() => {
  34     if (!flagOn) return;
  35     let live = true;
  36     void listPasskeys().then(result => { if (live) setLoad(result.ok ? { state: "ready", passkeys: result.value.passkeys ?? [] } : { state: "failed" }); });
  37     return () => { live = false; };
  38   }, [flagOn]);
  39   // Keep (the safe choice) takes focus when the confirmation opens.
  40   useEffect(() => {
  41     if (confirm) { dialog.current?.showModal(); dialog.current?.querySelector<HTMLElement>("[data-autofocus]")?.focus(); }
  42     else if (dialog.current?.open) dialog.current.close();
  43   }, [confirm]);
  44 
  45   if (!flagOn) return null;
  46 
  47   async function add() {
  48     if (busy) return;
  49     setBusy("add"); setNotice(null);
  50     const result = await registerPasskey(deviceName());
  51     setBusy(null);
  52     if (result.ok) { setNotice({ tone: "done", text: t.savedDone }); await refresh(); }
  53     else if (result.error !== "cancelled") setNotice({ tone: "error", text: t.setUpFailed });
  54   }
  55 
  56   async function remove(passkey: Passkey) {
  57     if (busy) return;
  58     setBusy("remove");
  59     const result = await removePasskey(passkey.id);
  60     setBusy(null); setConfirm(null);
  61     if (result.ok) { setNotice({ tone: "done", text: t.removedDone }); await refresh(); addButton.current?.focus(); }
  62     else { setNotice({ tone: "error", text: t.removeFailed }); opener.current?.focus(); }
  63   }
  64 
  65   const passkeys = load.state === "ready" ? load.passkeys : [];
  66   return <PasskeyDesktopView load={load.state} passkeys={passkeys} busy={busy} notice={notice} confirm={confirm} supported={supported} phone={phone} dialog={dialog} addButton={addButton}
  67     onRefresh={()=>void refresh()} onAdd={()=>void add()} onReview={(key,button)=>{opener.current=button;setConfirm(key);}} onRemove={key=>void remove(key)} onKeep={()=>{setConfirm(null);opener.current?.focus();}}/>;
  68 }
