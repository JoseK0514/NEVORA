  16 export function EmployeePasskeyView({mobile,state,supported,confirm,onAdd,onRefresh,onRemove,onReview,onKeep,onCancel,addRef,confirmationRef,feedbackRef}:ViewProps){
  17  const ready=state.phase==='ready',keys=ready?state.passkeys??[]:[],busy=state.busy!==null;
  18  const reviewed=ready?keys.find(key=>key.id===confirm?.id):null;
  19  if(state.phase==='disabled'||state.phase==='ended')return null;
  20  return <section className={`nv-card pk-section em-passkeys${mobile?' em-passkeys-mobile':''}`} aria-labelledby="passkeys-title" data-passkey-state={state.phase}>
  21   <div className="nv-card__head"><h2 className="nv-card__title" id="passkeys-title">{mobile?'Passkeys':'Face ID and passkeys'}</h2></div>
  22   <div className="nv-card__body pk-body">
  23    <p className="pk-muted em-passkeys-intro">Sign in with your device’s screen lock instead of typing your password.</p>
  24    {state.phase==='loading'?<p className="pk-muted" role="status">Checking your saved passkeys…</p>:null}
  25    {state.phase==='paused'&&!busy?<p className="pk-muted" role="status">Passkeys are paused. Return to this page to check your account.</p>:null}
  26    {state.phase==='denied'?<p className="pk-muted" role="status">Your session changed. <a href="/login">Sign in again</a> to manage your passkeys.</p>:null}
  27    {state.phase==='unavailable'?<p className="pk-muted">{state.needsRefresh?'Check your saved passkeys before starting another request.':'We couldn’t check your passkeys. Refresh to try again.'}</p>:null}
  28    {ready?<>
  29     {keys.length?<ul className="pk-list">{keys.map(key=><li className="pk-item" key={key.id}>
  30      <KeyIcon className="em-passkeys-key" size={24} aria-hidden="true"/>
  31      <span className="pk-item__text"><span className="pk-item__name">{key.name}</span><span className="pk-item__meta">Added {date(key.createdAt)}</span><span className="pk-item__meta">{key.lastUsedAt?'Last used '+date(key.lastUsedAt):'Not used yet'}</span></span>
  32      <button type="button" className="nv-link pk-remove" aria-label={'Remove '+key.name} disabled={busy||state.needsRefresh||!!reviewed} onClick={event=>onReview(key,event.currentTarget)}>Remove</button>
  33     </li>)}</ul>:<div className="em-passkeys-empty"><KeyIcon size={28} aria-hidden="true"/><div><h3>No saved passkeys</h3><p className="pk-muted">Your password still works. Add a passkey when you’re ready.</p></div></div>}
  34     {reviewed?<div className="em-passkeys-confirmation" role="group" aria-labelledby="passkey-remove-title" aria-describedby="passkey-remove-body" tabIndex={-1} ref={confirmationRef}>
  35      <h3 id="passkey-remove-title">Remove “{reviewed.name}”?</h3><p id="passkey-remove-body">This stops this passkey from signing in to NEVORA. It does not delete it from your device or sign you out.</p>
  36      {keys.length===1?<p>This is your last saved passkey. You can still sign in with your password.</p>:null}
  37      <div className="em-passkeys-actions"><button type="button" className="nv-btn" data-kind="secondary" disabled={busy} onClick={onKeep}>Keep passkey</button><button type="button" className="nv-btn" data-kind="danger" disabled={busy} onClick={()=>onRemove(reviewed.id)}>Remove passkey</button></div>
  38     </div>:null}
  39     {!reviewed&&supported?<button ref={addRef} type="button" className="nv-btn pk-add" data-kind="secondary" disabled={busy||state.needsRefresh} onClick={onAdd}><KeyIcon size={20} aria-hidden="true"/>Add a passkey</button>:null}
  40     {!supported?<p className="pk-muted">This browser can’t create a passkey. You can still manage your saved passkeys here, or use a supported browser to add one.</p>:null}

  43    <p className="em-passkeys-feedback" role="status" aria-live="polite" tabIndex={-1} ref={feedbackRef}>{state.notice}</p>
  44    {(!busy&&(state.phase==='unavailable'||state.phase==='paused'))?<button type="button" className="nv-btn" data-kind="secondary" onClick={onRefresh}>Refresh passkeys</button>:null}
  45    <details className="em-passkeys-help"><summary><span>How passkeys work</span><CaretDownIcon size={20} aria-hidden="true"/></summary><div><p>Your device may use a fingerprint, face scan, screen-lock code or security key.   50 /** Stable lifetime; responsive changes replace presentation, never a ceremony. */
  51 export function EmployeePasskeySection({authority}:{authority:string}){
  52  const mobile=useEmployeeMobile(),flag=passkeysFlagOn();
  53  const supported=useSyncExternalStore(subscribe,()=>passkeysAvailable()&&typeof navigator.credentials?.create==='function'&&window.isSecureContext,()=>false);
  54  const [state,setState]=useState<PasskeyState>({phase:'loading',busy:null,passkeys:null,notice:'',needsRefresh:false});
  55  const [confirm,setConfirm]=useState<Passkey|null>(null),owner=useRef<ReturnType<typeof createPasskeyManager>|null>(null);
  56  const add=useRef<HTMLButtonElement>(null),confirmation=useRef<HTMLDivElement>(null),feedback=useRef<HTMLParagraphElement>(null),opener=useRef<HTMLButtonElement|null>(null),previousBusy=useRef(state.busy);
  57  useEffect(()=>{
  58   if(!flag)return;
  59   const manager=createPasskeyManager(authority,()=>{if(manager.state.phase!=='ready')setConfirm(null);setState({...manager.state});});owner.current=manager;
  60   const resume=()=>{if(document.visibilityState!=='hidden'&&navigator.onLine!==false)void manager.resume();};
  61   const blur=()=>manager.pause(true),pause=()=>manager.pause(),end=()=>manager.end();
  62   const visibility=()=>document.visibilityState==='hidden'?pause():resume();
  63   const storage=(event:StorageEvent)=>{if(event.key===null||event.key==='fuel-portal-account-fence'||event.key==='fuel-portal-owner-generation')end();};
  64   void manager.refresh();window.addEventListener('blur',blur);window.addEventListener('pagehide',pause);window.addEventListener('offline',pause);window.addEventListener('focus',resume);window.addEventListener('pageshow',resume);window.addEventListener('online',resume);window.addEventListener('nevora:signout',end);window.addEventListener('storage',storage);document.addEventListener('visibilitychange',visibility);
  65   const poll=setInterval(()=>{if(document.visibilityState!=='hidden'&&manager.state.phase==='ready'&&!manager.state.busy)void manager.refresh();},30000);
  66   return()=>{clearInterval(poll);manager.end();owner.current=null;window.removeEventListener('blur',blur);window.removeEventListener('pagehide',pause);window.removeEventListener('offline',pause);window.removeEventListener('focus',resume);window.removeEventListener('pageshow',resume);window.removeEventListener('online',resume);window.removeEventListener('nevora:signout',end);window.removeEventListener('storage',storage);document.removeEventListener('visibilitychange',visibility);};
  67  },[authority,flag]);
  68  useEffect(()=>{if(confirm){confirmation.current?.focus();confirmation.current?.scrollIntoView({block:'nearest'});}},[confirm]);
  69  useEffect(()=>{const wasBusy=previousBusy.current;previousBusy.current=state.busy;if(wasBusy&&!state.busy&&state.notice){requestAnimationFrame(()=>{feedback.current?.focus();feedback.current?.scrollIntoView({block:'center'});});}},[state.busy,state.notice]);
  70  if(!flag)return null;
  71  return <EmployeePasskeyView mobile={mobile} state={state} supported={supported} confirm={confirm} addRef={add} confirmationRef={confirmation} feedbackRef={feedback}
  72   onAdd={()=>void owner.current?.add()} onRefresh={()=>void owner.current?.refresh()} onRemove={id=>void owner.current?.remove(id)} onCancel={()=>owner.current?.cancel()}
  73   onReview={(key,button)=>{opener.current=button;setConfirm(key);}} onKeep={()=>{setConfirm(null);requestAnimationFrame(()=>{if(opener.current?.isConnected)opener.current.focus();else add.current?.focus();});}}/>;
  74 }
