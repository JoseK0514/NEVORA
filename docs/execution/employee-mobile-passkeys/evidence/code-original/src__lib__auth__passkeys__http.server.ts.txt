   2 import { randomBytes } from "node:crypto";
   3 import { cookies } from "next/headers";
   4 import { getSession, requireSessionMutationProof, writeSessionCookie } from "../session";

   6 import { correctionJson, readCorrectionJson, correctionResponseHeaders } from "../../time-corrections/http.server";
   7 import { passkeyConfig } from "./config";
   8 import { PasskeyService } from "./service.server";
   9 import { PasskeyError, validateFlowToken } from "./types";
  10 import { employeeViewAuthority } from '../../employee-view-authority.server';

  12 type Action = "login_options" | "login_verify" | "register_options" | "register_verify" | "list" | "remove";
  13 const errorJson = (error: string, status: number) => correctionJson({ error }, status);
  14 const cookieName = (registration: boolean) => registration ? "nevora_pk_register" : "nevora_pk_login";
  15 function record(value: unknown): value is Record<string, unknown> {
  16   return !!value && typeof value === "object" && !Array.isArray(value);
  17 }
  18 function fields(value: unknown, expected: string[]): value is Record<string, unknown> {
  19   return record(value) && Object.keys(value).sort().join(",") === expected.sort().join(",");
  20 }

  22 /** Shared same-origin HTTP boundary for passkey ceremonies and management.
  23  * Separate HttpOnly flow cookies bind login/registration challenges; registration
  24  * and management additionally require the actual session mutation proof. Login
  25  * writes the normal session cookie only after the service commits its session. */
  26 export async function passkeyHttp(request: Request, action: Action, id?: string): Promise<Response> {
  27   const config = passkeyConfig(request.url);
  28   if (!config) return errorJson("not_supported", 404);
  29   const method = action === "list" ? "GET" : action === "remove" ? "DELETE" : "POST";
  30   if (request.method !== method) return errorJson("not_supported", 405);
  31   const site = request.headers.get("sec-fetch-site"), origin = request.headers.get("origin");
  32   if ((site && site !== "same-origin") || (origin && origin !== config.origin) || (method !== "GET" && origin !== config.origin))
  33     return errorJson("not_found", 403);
  34   if (new URL(request.url).search) return errorJson("not_found", 400);
  35   if (method === "POST" && request.body && request.headers.get("content-type")?.split(";")[0].trim().toLowerCase() !== "application/json")
  36     return errorJson("not_found", 403);
  37   try {
  38     const options = action.endsWith("_options");
  39     let body: unknown = {};
  40     if (method === "POST") {
  41       try { body = options && !request.body ? {} : await readCorrectionJson(request, 65536); }
  42       catch { return errorJson("not_found", 400); }
  43       if (!(options ? fields(body, []) : fields(body, action === "register_verify" ? ["response", "name"] : ["response"])))
  44         return errorJson("not_found", 400);
  45     }
  46     const protectedAction = action.startsWith("register") || action === "list" || action === "remove";
  47     let proof;
  48     if (protectedAction) {
  49       if (!(await getSession())) return errorJson("not_found", 401);
  50       try { proof = await requireSessionMutationProof(); } catch { return errorJson("not_found", 401); }
  51       const binding=request.headers.get('X-Nevora-Passkey-Authority');
  52       if(binding!==null&&(!['EMPLOYEE','LEAD'].includes(proof.employee.role)||binding!==employeeViewAuthority(proof,'passkeys')))
  53         return errorJson('not_found',401);
  54     }
  55     const service = new PasskeyService(config);
  56     if (action === "list" && proof) return request.headers.has('X-Nevora-Passkey-Authority')
  57       ?correctionJson({...await service.snapshot(proof),authority:employeeViewAuthority(proof,'passkeys')})
  58       :correctionJson({ passkeys: await service.list(proof) });
  59     if (action === "remove" && proof) {
  60       await service.remove(proof, id ?? "");
  61       return new Response(null, { status: 204, headers: correctionResponseHeaders });
  62     }
  63     const store = await cookies(), registration = action.startsWith("register");
  64     let token = store.get(cookieName(registration))?.value ?? "";
  65     if (options) {
  66       try { validateFlowToken(token); } catch { token = randomBytes(32).toString("base64url"); }
  67       const result = registration && proof
  68         ? await service.registrationOptions(proof, token) : await service.loginOptions(token);
  69       store.set(cookieName(registration), token, { httpOnly: true, secure: process.env.NODE_ENV === "production",
  70         sameSite: "strict", path: "/", maxAge: 900 });
  71       return correctionJson(result);
  72     }
  73     if (!record(body)) return errorJson("not_found", 400);
  74     if (action === "register_verify" && proof)
  75       return correctionJson({ passkey: await service.register(proof, token, body.response, body.name) });
  76     if (action === "login_verify") {
  77       const session = await service.login(token, body.response, async employeeId => { await prepareInstallationForLogin(employeeId); });
  78       await writeSessionCookie(session);
  79       return correctionJson({ redirect: "/app" });
  80     }
  81     return errorJson("not_found", 400);
  82   } catch (error) {
  83     if (error instanceof PasskeyError) return errorJson(error.code, error.code === "rate_limited" ? 429 :
  84       error.code === "account_disabled" ? 403 : error.code === "unavailable" ? 503 : 404);
  85     return errorJson("unavailable", 503);
  86   }
  87 }
