def redact(data, source):
    if source.suffix not in textual:
        return data, []
    try:
        text = data.decode('utf-8')
    except UnicodeDecodeError:
        return data, []
    reasons = []
    text, n = cookie.subn(r'\1[REDACTED_DISPOSABLE_FIXTURE_COOKIE]', text)
    if n: reasons.append({'kind':'disposable fixture cookie','occurrences':n})
    text, n = feed.subn(r'\1[REDACTED_DISPOSABLE_FEED]', text)
    if n: reasons.append({'kind':'disposable calendar feed capability','occurrences':n})
    if source.suffix == '.json':
        try:
            value = json.loads(text)
            secrets = set()
            def visit(obj):
                if isinstance(obj,dict):
                    if obj.get('name') in {'fuel_session','nevora_pk_register','nevora_pk_login'} and isinstance(obj.get('value'),str) and len(obj['value'])>=20:
                        secrets.add(obj['value'])
                    for child in obj.values(): visit(child)
                elif isinstance(obj,list):
                    for child in obj: visit(child)
            visit(value)
            for secret in secrets:
                if secret.startswith('[REDACTED'): continue
                text = text.replace(secret,'[REDACTED_DISPOSABLE_FIXTURE_COOKIE]')
                reasons.append({'kind':'disposable cookie record','occurrences':1})
        except (ValueError,TypeError):
            pass
    return text.encode('utf-8'), reasons

for root, label in roots:
    if not root.exists(): raise RuntimeError('Missing preservation source '+str(root))
    for parent, dirs, files in os.walk(root,followlinks=False):
        parent = Path(parent)
        kept = []
        for name in dirs:
            path = parent/name
            reason = None
            if path.is_symlink(): reason = 'symlink'
            elif name in {'node_modules','.next','.git','__pycache__','publish','owned-tmp','node-compile-cache'}: reason = 'runtime/dependency/generated or staging directory'
            elif name == 'app' and parent.name.startswith('owned-next-'): reason = 'verified runtime mirror; inputs/provenance preserved, source in Git'
            elif (path/'PG_VERSION').exists(): reason = 'database bytes'
            if reason: excluded.append({'source':str(path),'reason':reason})
            else: kept.append(name)
        dirs[:] = kept
        for name in files:
            source = parent/name
            if source.is_symlink() or not stat.S_ISREG(source.stat().st_mode):
                excluded.append({'source':str(source),'reason':'nonregular/symlink'}); continue
            if name.startswith('.env'):
                excluded.append({'source':str(source),'reason':'environment file'}); continue
            relative = Path(label)/source.relative_to(root)
            if source.suffix in code: relative = Path(str(relative)+'.txt')
            target = out/'evidence'/relative
            data = source.read_bytes()
            portable, redactions = redact(data,source)
            target.parent.mkdir(parents=True,exist_ok=True)
            target.write_bytes(portable)
            entry = {'source':str(source),'target':str(target.relative_to(out)),'sourceBytes':len(data),'portableBytes':len(portable),'sourceSHA256':digest(data),'portableSHA256':digest(portable),'redactions':redactions}
            if digest(target.read_bytes()) != entry['portableSHA256']: raise RuntimeError('Copy hash mismatch')
            entries.append(entry); by_source[str(source)] = entry
