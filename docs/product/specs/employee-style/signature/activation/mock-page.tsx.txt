// QA-only mock (throwaway preview copy, never committed): the invitation-based activation states in the Signature
// sign-in frame, with development's wording. /ui-preview/activate?state=ready|checking|missing|pending|mismatch|unavailable|unknown|activated
import { AUTH_COPY } from "@/lib/auth-copy";
import { AuthShell, WorkspaceEyebrow } from "@/components/auth-premium/auth-shell";
import { PasswordInput } from "@/components/auth-premium/password-input";
import { AuthDisclosureSymbol } from "@/components/auth-disclosure-symbol";
import { EnvelopeOpenIcon } from "@phosphor-icons/react/dist/ssr/EnvelopeOpen";
import { EnvelopeSimpleIcon } from "@phosphor-icons/react/dist/ssr/EnvelopeSimple";
import { CheckIcon } from "@phosphor-icons/react/dist/ssr/Check";
import { ArrowRightIcon } from "@phosphor-icons/react/dist/ssr/ArrowRight";
import { CopySimpleIcon } from "@phosphor-icons/react/dist/ssr/CopySimple";
import { WarningCircleIcon } from "@phosphor-icons/react/dist/ssr/WarningCircle";
import "../../login/auth-premium.css";
import "../../login/auth-signature.css";
import "./mock.css";

export const dynamic = "force-dynamic";

export default async function Page({ searchParams }: { searchParams: Promise<{ state?: string }> }) {
  const state = (await searchParams).state ?? "ready";
  const t = AUTH_COPY.en;
  const form = ["ready", "pending", "mismatch", "unavailable"].includes(state);
  const pending = state === "pending";
  const signIn = <a className="ap-link" href="/login">Already activated? Sign in</a>;
  const help = <details className="ap-help"><summary><span>No invitation yet?</span><AuthDisclosureSymbol /></summary>
    <p>Your workplace must create your employee record and send an invitation to your registered email. No invitation yet? Contact your Owner.</p></details>;

  const body = state === "activated" ? <>
    <span className="act-seal"><CheckIcon size={40} weight="bold" aria-hidden="true" /></span>
    <h1 className="ap-title">Account activated</h1>
    <p className="act-lead">Keep your Employee ID. You will use it with the password you just created.</p>
    <div className="act-id">
      <label className="act-id__label" htmlFor="act-id">Employee ID for sign-in</label>
      <div className="act-id__row"><input id="act-id" className="act-id__value" value="NS-10482" readOnly autoComplete="off" />
        <button type="button" className="act-id__copy"><CopySimpleIcon size={18} aria-hidden="true" />Copy</button></div>
    </div>
    <a className="ap-primary act-go" href="/login"><span>Continue to sign in</span><ArrowRightIcon size={20} aria-hidden="true" /></a>
  </> : <>
    <WorkspaceEyebrow name="Northside Services" />
    <h1 className="ap-title">Activate account</h1>
    {state === "missing" || state === "unknown" ? null : <p className="act-lead">Open your email invitation, then choose a password.</p>}

    {state === "checking" ? <div className="act-invite" role="status"><span className="act-invite__tile"><span className="act-spin" aria-hidden="true" /></span>
      <span className="act-invite__text"><b>Opening your invitation…</b><small>This takes a moment.</small></span></div> : null}
    {form ? <div className="act-invite"><span className="act-invite__tile"><EnvelopeOpenIcon size={24} weight="duotone" aria-hidden="true" /></span>
      <span className="act-invite__text"><b>Invitation opened</b><small>Choose a password to finish.</small></span>
      <CheckIcon className="act-invite__ok" size={20} weight="bold" aria-hidden="true" /></div> : null}

    {state === "missing" ? <div className="act-state">
      <span className="act-state__orb"><EnvelopeSimpleIcon size={30} weight="duotone" aria-hidden="true" /></span>
      <h2>Open the invitation link from your email to activate your account.</h2>
      <p>If the link cannot be used, ask your Owner for a new invitation. You can sign in if you have already activated.</p>
    </div> : null}
    {state === "unknown" ? <div className="act-state" data-tone="warn" role="alert">
      <span className="act-state__orb"><WarningCircleIcon size={30} weight="duotone" aria-hidden="true" /></span>
      <h2>Activation could not be confirmed.</h2>
      <p>Ask your Owner to confirm your account status and Employee ID. If you know your ID, try signing in before requesting a new invitation. Your account may already be activated.</p>
    </div> : null}

    {form ? <form className="ap-form act-form" aria-busy={pending}>
      <fieldset className="ap-fieldset" disabled={pending}>
        <div className="ap-field">
          <label className="ap-label" htmlFor="activation-password">Create password</label>
          <PasswordInput id="activation-password" name="password" minLength={8} placeholder="Choose your password" autoComplete="new-password" required
            defaultValue={state === "ready" ? "" : "correct-horse"} controls={["activation-password", "activation-confirm-password"]} showLabel="Show passwords" hideLabel="Hide passwords"
            aria-describedby="activation-password-hint" />
          <p className="ap-hint" id="activation-password-hint">At least 8 characters.</p>
        </div>
        <div className="ap-field" data-invalid={state === "mismatch" ? "" : undefined}>
          <label className="ap-label" htmlFor="activation-confirm-password">Confirm password</label>
          <div className="ap-box"><input id="activation-confirm-password" className="ap-input" name="confirmPassword" type="password" minLength={8}
            placeholder="Re-enter your password" autoComplete="new-password" required defaultValue={state === "ready" ? "" : state === "mismatch" ? "correct-horsf" : "correct-horse"}
            aria-invalid={state === "mismatch" || undefined} aria-describedby={state === "mismatch" || state === "unavailable" ? "activation-error" : undefined} /></div>
        </div>
        {state === "mismatch" || state === "unavailable" ? <p className="ap-alert" data-tone="error" id="activation-error" role="alert">
          {state === "mismatch" ? "Enter the same password in both fields." : "Activation is temporarily unavailable. Your invitation was not confirmed. Try again later."}</p> : null}
        <button type="submit" className="ap-primary" disabled={pending} aria-busy={pending}><span>{pending ? "Activating account…" : "Activate account"}</span></button>
      </fieldset>
    </form> : null}

    {state === "missing" || state === "unknown" ? <a className="ap-secondary act-signin" href="/login">Sign in</a> : null}
    <div className="ap-links act-links">{state === "missing" || state === "unknown" ? null : signIn}{help}</div>
  </>;
  return <AuthShell lang="en" t={t} variant="sub" back={state === "activated" ? undefined : "/login"}>{body}</AuthShell>;
}
