function read(base:string,path:string):Buffer {
  if(!path||path.startsWith('/')||path.includes('\\')||path.split('/').some(p=>!p||p==='.'||p==='..'))return fail();
  if(realpathSync(base)!==base||lstatSync(base).isSymbolicLink())return fail();
  let file=base;
  for(const part of path.split('/')){file=join(file,part);if(lstatSync(file).isSymbolicLink())return fail();}
  const distance=relative(base,file);if(distance.startsWith('..'+sep)||distance==='..')return fail();
  const fd=openSync(file,constants.O_RDONLY|constants.O_NOFOLLOW);
  try{const a=fstatSync(fd);if(!a.isFile())return fail();const result=readFileSync(fd),b=fstatSync(fd),c=lstatSync(file);
    if(a.dev!==b.dev||a.ino!==b.ino||a.size!==b.size||a.mtimeMs!==b.mtimeMs||a.ctimeMs!==b.ctimeMs||
      result.length!==b.size||c.isSymbolicLink()||c.dev!==b.dev||c.ino!==b.ino)return fail();return result;

function installedPaths(base:string,directory:string):string[] {

/** Closed bootstrap, not request-time arbitrary loader. Only admitted after root
 * supplies controlled generated installation enrollment and immutable runtime
 * installation/import custody. Equality cannot authenticate the producer. */
export async function loadPortalBuildInstallStartupIdentity() {
  try {
    if(process.env.NODE_ENV!=='production'||process.env.NODE_OPTIONS||process.env.NODE_PATH||process.env.__NEXT_PRIVATE_STANDALONE_CONFIG||process.execArgv.length)return denied;
    const root=resolve(process.cwd()),buildIdBytes=read(root,'.next/BUILD_ID'),buildId=buildIdBytes.toString('utf8');
    if(!buildId||/[\r\n]/.test(buildId)||!buildIdBytes.equals(Buffer.from(buildId,'utf8')))return denied;
    // Fixed sibling produced by the opt-in source extractor. No env path/current
    // pointer; native BUILD_ID resolves a create-only immutable record.
    const retainRoot=resolve(root,'../portal-build-bindings');
    const stat=lstatSync(retainRoot);if(!stat.isDirectory()||stat.isSymbolicLink()||stat.uid!==process.getuid?.()||(stat.mode&0o077)!==0)return denied;
    const slot=exact(json(read(retainRoot,'by-build-id/'+hash(buildIdBytes)+'.json')),['format','buildId','bindingDigest']);
    if(slot.format!=='PORTAL_BUILD_BINDING_SLOT_V1'||slot.buildId!==buildId||typeof slot.bindingDigest!=='string'||!/^[a-f0-9]{64}$/.test(slot.bindingDigest))return denied;
    const retainedBytes=read(retainRoot,slot.bindingDigest+'/binding.json');if(hash(retainedBytes)!==slot.bindingDigest)return denied;
    const retained=exact(json(retainedBytes),['format','checkpoint','tree','releaseReceipt','runtime','buildId','sourceHashes','artifactHashes',
      'requiredFiles','entries','traceEdges','dependencies','image','launcher','installationBytes','generatedImageSourceSha256','installationSourceSha256','verification','remainingGates']);
    if(retained.format!=='PORTAL_BUILD_INSTALL_INPUT_RETENTION_V1'||retained.buildId!==buildId)return denied;
    const runtime=exact(retained.runtime,['node','npm','platform','architecture','next','packageSha256','lockSha256']);
    if(runtime.node!==process.version||runtime.platform!==process.platform||runtime.architecture!==process.arch||runtime.next!=='16.4.2')return denied;
    const launcher=exact(retained.launcher,['path','sha256','command','nodeOptions','nodePath','standaloneConfig']);
    if(launcher.path!=='node_modules/next/dist/bin/next'||launcher.command!=='next start'||launcher.nodeOptions!=='ABSENT'||launcher.nodePath!=='ABSENT'||launcher.standaloneConfig!=='ABSENT'||
      process.argv[2]!=='start'||!process.argv[1]||realpathSync(process.argv[1])!==join(root,'node_modules/next/dist/bin/next')||hash(read(root,launcher.path))!==launcher.sha256)return denied;
    const hashes=record(retained.artifactHashes);
    if(JSON.stringify(installedPaths(root,'.next'))!==JSON.stringify(Object.keys(hashes).sort()))return denied;
    for(const [path,digest] of Object.entries(hashes))if(!path.startsWith('.next/')||typeof digest!=='string'||hash(read(root,path))!==digest)return denied;
    if(!Array.isArray(retained.dependencies))return denied;
    for(const entry of retained.dependencies){const dependency=exact(entry,['path','sha256','byteLength']);

    if(hash(read(root,'package.json'))!==runtime.packageSha256||hash(read(root,'package-lock.json'))!==runtime.lockSha256)return denied;

        'CONTROLLED_INSTALLER_SOURCE_AND_OPERATOR_ENROLLMENT','IMMUTABLE_ACTUAL_SERVER_IMPORT_INSTALLATION_CUSTODY',
        'ACTUAL_NATIVE_PRIVATE_ENROLLMENT_CAPSULE_AND_FINAL_SOURCE_QA','PRIVILEGED_WRITER_FENCE_ADOPTION',
        'PROTECTED_RESOLVER_FULL_SCOPE_CURRENT_PROOF_PRIVATE_RLS_RECOVERY_NEON'])});

  }catch{return denied;}
}
